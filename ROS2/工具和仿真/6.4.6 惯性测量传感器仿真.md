# gazebo_sensor_plugin.xacro代码
```xacro
       <!-- IMU惯性测量单元传感器配置 -->
        <gazebo reference="imu_link">
            <!-- IMU传感器定义 -->
            <sensor name="imu_sensor" type="imu">
                <!-- 发布话题名称 -->
                <topic>imu</topic>
                <!-- 更新频率 100Hz -->
                <update_rate>100</update_rate>
                <!-- 始终保持开启状态 -->
                <always_on>1</always_on>
                <!-- 是否在Gazebo中可视化 -->
                <visualize>true</visualize>
                <!-- 六轴噪声设置 -->
                <imu>
                    <!-- 角速度传感器噪声配置 (陀螺仪) -->
                    <angular_velocity>
                        <!-- X轴角速度噪声 -->
                        <x>
                            <!-- 高斯噪声模型 -->
                            <noise type="gaussian">
                                <!-- 噪声均值 -->
                                <mean>0.0</mean>
                                <!-- 噪声标准差 -->
                                <stddev>2e-4</stddev>
                                <!-- 偏差均值 -->
                                <bias_mean>0.0000075</bias_mean>
                                <!-- 偏差标准差 -->
                                <bias_stddev>0.0000008</bias_stddev>
                            </noise>
                        </x>
                        <!-- Y轴角速度噪声 -->
                        <y>
                            <noise type="gaussian">
                                <mean>0.0</mean>
                                <stddev>2e-4</stddev>
                                <bias_mean>0.0000075</bias_mean>
                                <bias_stddev>0.0000008</bias_stddev>
                            </noise>
                        </y>
                        <!-- Z轴角速度噪声 -->
                        <z>
                            <noise type="gaussian">
                                <mean>0.0</mean>
                                <stddev>2e-4</stddev>
                                <bias_mean>0.0000075</bias_mean>
                                <bias_stddev>0.0000008</bias_stddev>
                            </noise>
                        </z>
                    </angular_velocity>
                    <!-- 线性加速度传感器噪声配置 (加速度计) -->
                    <linear_acceleration>
                        <!-- X轴线性加速度噪声 -->
                        <x>
                            <noise type="gaussian">
                                <mean>0.0</mean>
                                <stddev>1.7e-2</stddev>
                                <bias_mean>0.1</bias_mean>
                                <bias_stddev>0.001</bias_stddev>
                            </noise>
                        </x>
                        <!-- Y轴线性加速度噪声 -->
                        <y>
                            <noise type="gaussian">
                                <mean>0.0</mean>
                                <stddev>1.7e-2</stddev>
                                <bias_mean>0.1</bias_mean>
                                <bias_stddev>0.001</bias_stddev>
                            </noise>
                        </y>
                        <!-- Z轴线性加速度噪声 -->
                        <z>
                            <noise type="gaussian">
                                <mean>0.0</mean>
                                <stddev>1.7e-2</stddev>
                                <bias_mean>0.1</bias_mean>
                                <bias_stddev>0.001</bias_stddev>
                            </noise>
                        </z>
                    </linear_acceleration>
                </imu>
            </sensor>
```
