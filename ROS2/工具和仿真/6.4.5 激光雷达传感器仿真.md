# gazebo_sensor_plugin.xacro代码
```xacro
<?xml version="1.0"?>
<robot xmlns:xacro="http://www.ros.org/wiki/xacro">
    <!-- Gazebo传感器插件宏定义 -->
    <xacro:macro name="gazebo_sensor_plugin">
        <!-- 激光雷达传感器配置 -->
        <gazebo reference="laser_link">
            <!-- GPU加速的激光雷达传感器 -->
            <sensor name="gpu_lidar" type="gpu_lidar">
                <!-- 发布话题名称 -->
                <topic>lidar</topic>
                <!-- 更新频率 100Hz -->
                <update_rate>100</update_rate>
                <lidar>
                    <!-- 设置扫描范围 -->
                    <scan>
                        <!-- 描述水平扫描范围 -->
                        <horizontal>
                            <!-- 每圈采样光束数量 -->
                            <samples>360</samples>
                            <!-- 分辨率，与samples一起决定实际采样点数量 -->
                            <resolution>1.000000</resolution>
                            <!-- 起始扫描角度 (弧度) -->
                            <min_angle>0.000000</min_angle>
                            <!-- 终止扫描角度 (弧度，约等于2π) -->
                            <max_angle>6.283100</max_angle>
                        </horizontal>
                    </scan>
                    <!-- 设置扫描距离 -->
                    <range>
                        <!-- 最小检测距离 (米) -->
                        <min>0.080000</min>
                        <!-- 最大检测距离 (米) -->
                        <max>10.0</max>
                        <!-- 距离分辨率 (米) -->
                        <resolution>0.010000</resolution>
                    </range>
                </lidar>
                <!-- 是否在Gazebo中可视化激光束 -->
                <visualize>true</visualize>
            </sensor>
        </gazebo>
   </xacro:macro>
</robot>

```
