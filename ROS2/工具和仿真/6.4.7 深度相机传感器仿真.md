```xacro
<?xml version="1.0"?>
<robot xmlns:xacro="http://www.ros.org/wiki/xacro">
    <xacro:include filename="$(find szcbot_description)/urdf/szcbot/common_inertia.xacro"/>


    <!-- 机器人的摄像头部件 -->
        <xacro:macro name="camera_xacro" params="xyz">
        <link name="camera_link">
        <!-- 部件的外观描述 -->
        <visual>
        <!-- 沿着自己几何中心的偏移和旋转 -->
            <origin xyz="0.0 0.0 0.0" rpy="0.0 0.0 0.0"/>
            <!-- 几何形状 -->
            <geometry>
                <box size="0.02 0.10 0.02"/>
            </geometry>
            <!-- 材质和颜色 -->
            <material name="black">
                <color rgba="0.0 0.0 0.0 0.5"/>
            </material>
        </visual>
        <collision>
        <!-- 沿着自己几何中心的偏移和旋转 -->
            <origin xyz="0.0 0.0 0.0" rpy="0.0 0.0 0.0"/>
            <!-- 几何形状 -->
            <geometry>
                <box size="0.02 0.10 0.02"/>
            </geometry>
            <!-- 材质和颜色 -->
            <material name="black">
                <color rgba="0.0 0.0 0.0 0.5"/>
            </material>
        </collision>
        <xacro:box_inertia m="0.1" w="0.02" h="0.10" d="0.02" />

    </link>

        <xacro:macro name="camera_xacro" params="xyz">
        <link name="camera_optical_link">


    </link>

        <!-- 机器人的关节，用于组合各个部件 -->
    <joint name="camera_joint" type="fixed">
        <!-- 关节的父部件 -->
        <parent link="base_link"/>
        <!-- 关节的子部件 -->
        <child link="camera_link"/>
        <!-- 关节的偏移和旋转 -->
        <origin xyz="${xyz}" rpy="0.0 0.0 0.0"/>
    </joint>

    <joint name="camera_optical_link_joint" type="fixed">
        <!-- 关节的父部件 -->
        <parent link="camera_link"/>
        <!-- 关节的子部件 -->
        <child link="camera_optical_link"/>
        <!-- 关节的偏移和旋转 -->
        <origin xyz="0.0 0.0 0.0" rpy="${-pi/2} 0.0 0.0"/>
    </joint>
            <gazebo reference ="camera_link">
        <material>Gazebo/Black</material>
    </gazebo>
    </xacro:macro>

</robot>
```

# gazebo_sensor_plugin.xacro代码
```xacro
              <!-- RGB相机传感器配置 -->
        <gazebo reference="camera_link">
            <!-- RGB相机传感器定义 -->
            <sensor name="camera" type="camera">
                <!-- 传感器位姿 (x y z roll pitch yaw) -->
                <pose>0 0 0 0 0 0</pose>
                <!-- 发布话题名称 -->
                <topic>/camera</topic>
                <!-- 更新频率 25Hz -->
                <update_rate>25</update_rate>
                <!-- 是否启用性能指标统计 -->
                <enable_metrics>false</enable_metrics>
                <!-- 相机参数配置 -->
                <camera name="head">
                    <!-- 相机位姿 -->
                    <pose>0 0 0 0 0 0</pose>
                    <!-- 水平视场角 (弧度) -->
                    <horizontal_fov>1.3</horizontal_fov>
                    <!-- 图像参数 -->
                    <image>
                        <!-- 图像宽度 (像素) -->
                        <width>640</width>
                        <!-- 图像高度 (像素) -->
                        <height>480</height>
                        <!-- 图像格式 -->
                        <format>RGB_INT8</format>
                        <!-- 抗锯齿采样倍数 -->
                        <anti_aliasing>4</anti_aliasing>
                    </image>
                    <!-- 相机信息话题名称，使用默认值 -->
                    <camera_info_topic>__default__</camera_info_topic>
                    <!-- 触发话题，空表示不使用触发模式 -->
                    <trigger_topic></trigger_topic>
                    <!-- 是否使用触发模式 -->
                    <triggered>false</triggered>
                    <!-- 相机裁剪平面 -->
                    <clip>
                        <!-- 近裁剪平面距离 (米) -->
                        <near>0.02</near>
                        <!-- 远裁剪平面距离 (米) -->
                        <far>30</far>
                    </clip>
                    <!-- 可见性掩码，用于选择性渲染 -->
                    <visibility_mask>4294967295</visibility_mask>
                    <!-- 图像噪声模型 -->
                    <noise>

                        <type>gaussian</type>
                        <!-- 噪声均值 -->
                        <mean>0</mean>
                        <!-- 噪声标准差 -->
                        <stddev>0.0070000000000000001</stddev>
                    </noise>
                </camera>
            </sensor>
        </gazebo>
```
