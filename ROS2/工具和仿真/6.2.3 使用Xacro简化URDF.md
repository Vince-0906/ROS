# xacro代码
```xacro
<?xml version="1.0"?>
<robot xmlns:xacro="http://www.ros.org/wiki/xacro" name="first_robot">

    <xacro:macro name="base_link" params="length radius">
    <!-- 机器人的身体部分 -->
    <link name="base_link">
        <!-- 部件的外观描述 -->
        <visual>
        <!-- 沿着自己几何中心的偏移和旋转 -->
            <origin xyz="0.0 0.0 0.0" rpy="0.0 0.0 0.0"/>
            <!-- 几何形状 -->
            <geometry>
                <!-- 圆柱体 半径0.10m，高度0.12m -->
                <cylinder radius="${radius}" length="${length}"/>
            </geometry>
            <!-- 材质和颜色 -->
            <material name="white">
                <color rgba="1.0 1.0 1.0 0.5"/>
            </material>
        </visual>
    </link>
    </xacro:macro>



    <!-- 机器人的IMU部件，惯性测量传感器 -->
        <xacro:macro name="imu_link" params="imu_name xyz">
        <link name="${imu_name}_link">
        <!-- 部件的外观描述 -->
        <visual>
        <!-- 沿着自己几何中心的偏移和旋转 -->
            <origin xyz="0.0 0.0 0.0" rpy="0.0 0.0 0.0"/>
            <!-- 几何形状 -->
            <geometry>
                <!-- 正方体0.02m x 0.02m x 0.02m -->
                <box size="0.02 0.02 0.02"/>
            </geometry>
            <!-- 材质和颜色 -->
            <material name="black">
                <color rgba="0.0 0.0 0.0 0.5"/>
            </material>
        </visual>
    </link>
        <!-- 机器人的关节，用于组合各个部件 -->
    <joint name="${imu_name}_joint" type="fixed">
        <!-- 关节的父部件 -->
        <parent link="base_link"/>
        <!-- 关节的子部件 -->
        <child link="${imu_name}_link"/>
        <!-- 关节的偏移和旋转 -->
        <origin xyz="${xyz}" rpy="0.0 0.0 0.0"/>
    </joint>
    </xacro:macro>

<xacro:base_link length="0.12" radius="0.10"/>
<xacro:imu_link imu_name="imu" xyz="0.0 0.0 0.06"/>
<xacro:imu_link imu_name="imu_down" xyz="0.0 0.0 -0.06"/>

</robot>
```
需要注意的就是不管是`joint`还是`name`都得用`${imu_name}`，不然会报错，因为用了同一个名字
