# 传感器
## imu
```xacro
<?xml version="1.0"?>
<robot xmlns:xacro="http://www.ros.org/wiki/xacro">

    <!-- 机器人的IMU部件，惯性测量传感器 -->
        <xacro:macro name="imu_xacro" params="xyz">
        <link name="imu_link">
        <!-- 部件的外观描述 -->
        <visual>
        <!-- 沿着自己几何中心的偏移和旋转 -->
            <origin xyz="0.0 0.0 0.0" rpy="0.0 0.0 0.0"/>
            <!-- 几何形状 -->
            <geometry>
                <!-- 正方体0.02m x 0.02m x 0.02m -->
                <box size="0.02 0.02 0.02"/>
            </geometry>
            <!-- 材质和颜色 -->
            <material name="black">
                <color rgba="0.0 0.0 0.0 0.5"/>
            </material>
        </visual>
    </link>
        <!-- 机器人的关节，用于组合各个部件 -->
    <joint name="imu_joint" type="fixed">
        <!-- 关节的父部件 -->
        <parent link="base_link"/>
        <!-- 关节的子部件 -->
        <child link="imu_link"/>
        <!-- 关节的偏移和旋转 -->
        <origin xyz="${xyz}" rpy="0.0 0.0 0.0"/>
    </joint>
    </xacro:macro>

</robot>
```

# 主体
## base
```xacro
<?xml version="1.0"?>
<robot xmlns:xacro="http://www.ros.org/wiki/xacro">

    <xacro:macro name="base_xacro" params="length radius">
    <!-- 机器人的身体部分 -->
    <link name="base_link">
        <!-- 部件的外观描述 -->
        <visual>
        <!-- 沿着自己几何中心的偏移和旋转 -->
            <origin xyz="0.0 0.0 0.0" rpy="0.0 0.0 0.0"/>
            <!-- 几何形状 -->
            <geometry>
                <!-- 圆柱体 半径0.10m，高度0.12m -->
                <cylinder radius="${radius}" length="${length}"/>
            </geometry>
            <!-- 材质和颜色 -->
            <material name="white">
                <color rgba="1.0 1.0 1.0 0.5"/>
            </material>
        </visual>
    </link>
    </xacro:macro>

</robot>
```

## szcbot
```xacro
<?xml version="1.0"?>
<robot xmlns:xacro="http://www.ros.org/wiki/xacro" name="szcbot_robot">
<!-- 身体 -->
<xacro:include filename="$(find szcbot_description)/urdf/szcbot/base.urdf.xacro"/>
<!-- 传感器 -->
<xacro:include filename="$(find szcbot_description)/urdf/szcbot/sensor/laser.urdf.xacro"/>
<xacro:include filename="$(find szcbot_description)/urdf/szcbot/sensor/imu.urdf.xacro"/>
<xacro:include filename="$(find szcbot_description)/urdf/szcbot/sensor/camera.urdf.xacro"/>

<xacro:base_xacro length="0.12" radius="0.10"/>
<xacro:imu_xacro xyz="0.0 0.0 0.02"/>
<xacro:camera_xacro xyz="0.10 0.0 0.075"/>
<xacro:laser_xacro  xyz="0.00 0.0 0.11"/>


</robot>

```
